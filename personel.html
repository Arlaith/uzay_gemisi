OKSİJEN HESAPLAYICI (Switch-Case ve Return Kullanımı) // Ana fonksiyon function oksijenHesapla() { var rutbe = document.getElementById("rutbeSecim").value; // Yardımcı fonksiyona rütbeyi gönderip, dönen (return) sonucu alıyoruz. var harcananLitre = litreBul(rutbe); // Sonucu Ekrana Yazdırma document.getElementById("sonucAlani").innerHTML = "Günlük Tüketim: " + harcananLitre + " Litre"; } // Geriye Değer Döndüren Yardımcı Fonksiyon function litreBul(gelenRutbe) { var litre = 0; // Switch-Case yapısı: Bir değişkenin çok fazla ihtimali varsa If-Else yerine kullanılır. switch (gelenRutbe) { case "kaptan": litre = 450; // Kaptan stresli, çok nefes alıyor :) break; // İşlem bitince bloktan çık! case "muhendis": litre = 600; // Efor sarf ediyor break; case "bilim": litre = 300; // Sakin sakin oturuyor break; default: litre = 0; // Hiçbiri değilse } // Hesaplanan değeri asıl fonksiyona GERİ DÖNDÜR (Return) return litre; }
